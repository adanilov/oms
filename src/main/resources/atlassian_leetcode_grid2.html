class StockPrice {

    Map<Integer, Integer>  priceByTs = new ConcurrentHashMap<>();
    TreeMap<Integer, Integer> freqByPrice = new TreeMap<>();
    int maxTs = -1;

    public void update(int timestamp, int price) {
        maxTs = Math.max(maxTs, timestamp);

        Integer oldPrice = priceByTs.get(timestamp);
        if(oldPrice != null) {
            Integer freq = freqByPrice.get(oldPrice);
            if (freq == 1) {
                freqByPrice.remove(oldPrice);
            } else {
                freqByPrice.put(oldPrice, freq - 1);
            }
        }

        priceByTs.put(timestamp, price);

        Integer freq = freqByPrice.get(price);
        if (freq == null) {
            freqByPrice.put(price, 1);
        } else {
            freqByPrice.put(price, freq + 1);
        }
    }

    public int current() {
        return priceByTs.get(maxTs);
    }

    public int maximum() {
        return freqByPrice.lastKey();
    }

    public int minimum() {
        return freqByPrice.firstKey();
    }
  }


class Solution {
    public List<String> findHighAccessEmployees(List<List<String>> access_times) {
        /*
            get for 1 employee
            sort
            find out
        */
        Map<String, List<String>> timesByEmployee = new HashMap<>();
        for(List<String> rec : access_times) {
            String empl = rec.get(0);
            String time = rec.get(1);
            timesByEmployee.putIfAbsent(empl, new ArrayList<>());
            timesByEmployee.get(empl).add(time);
        }

        List<String> res = new ArrayList<>();

        for(var entry:timesByEmployee.entrySet()) {
            List<String> times = entry.getValue();
            Collections.sort(times);
            for(int i=1; i<times.size()-1;i++){
                if( diff(times.get(i-1), times.get(i)) + diff(times.get(i), times.get(i+1)) <60 ) {
                    res.add(entry.getKey());
                    break;
                }
            }
        }
        return res;
    }
    int diff(String a, String b) {
        int hoursA = Integer.valueOf(a.substring(0,2));
        int hoursB = Integer.valueOf(b.substring(0,2));

        int minsA = Integer.valueOf(a.substring(2));
        int minsB = Integer.valueOf(b.substring(2));
        int res = (hoursB - hoursA)*60;
        if (minsB<minsA){
            minsB += 60;
            res -= 60;
        }
        res += minsB - minsA;
        //System.out.println("between "+a+" and "+b+" " +res + " mins");
        return res;
    }
}


class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        ''' Approach:
        for each cell, if it is 1st letter ot the word:
            dfs
                pass: next letter to look for
                    set of visited
                    dfs returns what? True/False
        '''
        ROWS, COLS = len(board), len(board[0])

        def dfs(r,c, i, visited):
            if i == len(word):
                return True

            if r < 0 or r >= ROWS or c < 0 or c >= COLS or (r,c) in visited or word[i] != board[r][c]:
                return False

            visited.add((r,c))
            if (dfs(r+1,c,i+1,visited)
                or dfs(r-1,c,i+1,visited)
                or dfs(r,c+1, i+1, visited)
                or dfs(r,c-1, i+1, visited)):
                return True
            visited.remove((r,c))

        for r in range(ROWS):
            for c in range(COLS):
                if board[r][c] == word[0]:
                    if dfs(r, c, 0, set()):
                        return True
        return False


class Solution {
    int ROWS, COLS;
    int[][] matrix;

    boolean dfs(int r, int c, int[][] visited) {//true = > canCross
        if (r<0 || r >= ROWS || c<0 || c >= COLS) {
            return false;
        }

        if(visited[r][c]==1) {
            return false;
        }
        if(matrix[r][c]==1) {
            return false;
        }
        visited[r][c] = 1;
        if(r == ROWS - 1) {
            return true;
        }
        if(dfs(r-1,c,visited)
            || dfs(r+1,c,visited)
            || dfs(r,c-1,visited)
            || dfs(r,c+1,visited)
        ) {
            return true;
        }
        return false;
    }

    public int latestDayToCross(int row, int col, int[][] cells) {
        this.ROWS = row;
        this.COLS = col;
        int l = 0;
        int r = cells.length - 1;
        int res = 0;
        while(l <= r) {
            int day = (l+r) / 2;
            int[][] curMatrix = new int[ROWS][COLS];
            for (int i=0; i < day; i++) {
                int[] water = cells[i];
                curMatrix[water[0]-1][water[1]-1] = 1;
            }
            this.matrix = curMatrix;
            boolean canCross = false;

            for(int c=0;c < COLS;c++){
                canCross = dfs(0, c, new int[ROWS][COLS]);

                if(canCross) {
                    break;
                }
            }
            if(canCross) {
                res = Math.max(res, day);
                l = day + 1;
            } else {
                r = day - 1;
            }
        }
        return res;
    }

}


public Node lowestCommonAncestor(Node p, Node q) {
    /*
        get distance for both
        advance the lower one by the diff (so they're both at one level)
        advance node=node.parent until they point to the same node
    */
    Node pCopy = p;
    Node qCopy = q;

    int pLen = 0;
    while (p!=null) {
        p = p.parent;
        pLen++;
    }

    int qLen = 0;
    while (q!=null) {
        q = q.parent;
        qLen++;
    }

    p = pCopy;
    q = qCopy;
    while (pLen>qLen) {
        p = p.parent;
        pLen--;
    }
    while (qLen>pLen){
        q = q.parent;
        qLen--;
    }

    while(p!=q){
        p = p.parent;
        q = q.parent;
    }
    return p;

}


Map<String, Integer> tsByMessage = new HashMap<>();

public boolean shouldPrintMessage(int timestamp, String message) {
    Integer lastTs = tsByMessage.get(message);
    if(lastTs==null){
        tsByMessage.put(message, timestamp);
        return true;
    }
    if(timestamp - lastTs >= 10) {
        tsByMessage.put(message, timestamp);
        return true;
    } else {
        return false;
    }
}


public int numPairsDivisibleBy60(int[] time) {
    int[] rems = new int[60];
    int res = 0;
    for(int t : time) {
        t %= 60;
        if (t != 0) {
            res += rems[60-t];
        } else {
            res += rems[0];
        }
        rems[t]++;
    }
    return res;
}

class FoodRatings {

    /*
    in cuisine, we need a sorted collection (by rating dec, then by name inc)
    Heap: miso, 1; sushi, 1; sorted.
    map: cuisine by food. We need it to locate the right heap.
    */
    static class FoodAndRating implements Comparable<FoodAndRating> {
        String food;
        int rating;
        public FoodAndRating(String food, int rating) {this.food = food; this.rating = rating;}
        public int compareTo(FoodAndRating that) {
            if (this.rating == that.rating){
                return this.food.compareTo(that.food);
            } else {
                return new Integer(that.rating).compareTo(this.rating);
            }
        }
    }

    Map<String, String> cuisineByFood = new HashMap<>();
    Map<String, PriorityQueue<FoodAndRating>> ratingsByCuisine = new HashMap<>();
    Map<String, Integer> ratingByFood = new HashMap<>();

    public FoodRatings(String[] foods, String[] cuisines, int[] ratings) {
        for(int i=0;i < foods.length;i++){
            String food = foods[i];
            String cuisine = cuisines[i];
            int rating = ratings[i];
            var far = new FoodAndRating(food, rating);
            cuisineByFood.put(food, cuisine);
            ratingsByCuisine.putIfAbsent(cuisine, new PriorityQueue<>());
            ratingsByCuisine.get(cuisine).add(far);
            ratingByFood.put(food, rating);
        }
    }

    public void changeRating(String food, int newRating) {
        var heap = ratingsByCuisine.get(cuisineByFood.get(food));
        heap.add(new FoodAndRating(food, newRating));
        ratingByFood.put(food,newRating);
    }

    public String highestRated(String cuisine) {
        var heap = ratingsByCuisine.get(cuisine);
        while(true) {
            FoodAndRating far = heap.peek();
            if (far.rating != ratingByFood.get(far.food)) {
                heap.poll();
            } else {
                break;
            }
        }
        return heap.peek().food;
    }
}


public int[] vowelStrings(String[] words, int[][] queries) {
    /*
        int[] prefix - each element is the number of strings before i including i starting and ending with vowel
        1 1 2
        return the diff
    */
    Set vowels = Set.of('a','o', 'u', 'i', 'e');
    int[] pref = new int[words.length];
    int vowelWordsSoFar = 0;
    for(int i=0; i< words.length; i++) {
        String w = words[i];
        if (vowels.contains(w.charAt(0)) && vowels.contains(w.charAt(w.length()-1)) ) {
            vowelWordsSoFar++;
        }
        pref[i] = vowelWordsSoFar;
    }
    int[] res = new int[queries.length];
    for (int i = 0; i < queries.length; i++) {
        int[] q = queries[i];
        res[i] = q[0] == 0 ? pref[q[1]] : pref[q[1]] - pref[q[0]-1];
    }
    return res;
}
