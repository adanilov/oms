public static int countPalindromes(String s) {
    int len = s.length();
    int[] count = new int[10];
    int[][][] pref = new int[len][10][10];
    int[][][] suff = new int[len][10][10];

    for (int i = 0; i < len; i++) {
        int dig = s.charAt(i) - '0';

        if (i > 0) {
            for (int j = 0; j < 10; j++) {
                for (int k = 0; k < 10; k++) {
                    pref[i][j][k] = pref[i - 1][j][k];
                    if (k == dig) {
                        pref[i][j][k] += count[j];
                    }
                }
            }
        }
        count[dig]++;
    }

    Arrays.fill(count, 0);

    for (int i = len - 1; i >= 0; i--) {
        int dig = s.charAt(i) - '0';

        if (i < len - 1) {
            for (int j = 0; j < 10; j++) {
                for (int k = 0; k < 10; k++) {
                    suff[i][j][k] = suff[i + 1][j][k];
                    if (dig == k) {
                        suff[i][j][k] += count[j];
                    }

                }
            }
        }
        count[dig]++;
    }
    int res = 0;
    for (int i = 1; i < len - 1; i++) {
        for (int j = 0; j < 10; j++) {
            for (int k = 0; k < 10; k++) {
                res += pref[i - 1][j][k] * suff[i + 1][j][k];
            }
        }
    }
    return res;
}
