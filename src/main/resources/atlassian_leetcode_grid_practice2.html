public String rankTeams(String[] votes) {
/*
    1)create TEAM:SortedListOfRanks objects
    A:11111
    B:22233
    C:22333
    2)We need a comparator comparing two sorted lists of Integers
    3)when sorted, return TEAMs
*/
Map<Character, List<Integer>> votesByTeam = new HashMap<>();
for (String v : votes) {
    for (int rank = 0; rank < v.length(); rank++) {
        char team = v.charAt(rank);
        votesByTeam.putIfAbsent(team, new ArrayList<>());
        votesByTeam.get(team).add(rank);
    }
}

class TeamAndVotes implements Comparable<TeamAndVotes> {
    List<Integer> votes;
    Character team;
    public int compareTo(TeamAndVotes that) {
        int i = 0;
        while(i<this.votes.size() && i < that.votes.size()) {
            if(this.votes.get(i) < that.votes.get(i)){
                return -1;
            } else if(this.votes.get(i) > that.votes.get(i)){
                return 1;
            } else {
                i++;
            }
        }
        return this.team.compareTo(that.team);
    }
}
List<TeamAndVotes> list = new ArrayList<>();
for (var entry: votesByTeam.entrySet()) {
    Collections.sort(entry.getValue());
    TeamAndVotes tav = new TeamAndVotes();
    tav.team = entry.getKey();
    tav.votes = entry.getValue();
    list.add(tav);
}

Collections.sort(list);
StringBuilder res = new StringBuilder();
for (var tav:list) {
    res.append(tav.team);
}
return res.toString();
}

public int maxProfit(int[] prices) {

    int maxProfit = 0;
    int minPrice = Integer.MAX_VALUE;

    for (int price : prices) {
        int profit = price - minPrice;
        if(profit > maxProfit)
            maxProfit = profit;
        if(price < minPrice)
            minPrice = price;
    }
    return maxProfit;
}

def minMeetingRooms(self, intervals: List[List[int]]) -> int:
    delta = []
    for a,b in intervals:
        delta.append((a, 1))
        delta.append((b, -1))
    delta.sort()
    cur = 0
    res = 0
    for k,v in delta:
        cur += v
        res = max(res, cur)
    return res


def merge(self, intervals: List[List[int]]) -> List[List[int]]:
    ''' Approach:
    sort by start
    cur, next
    1 if next.start > cur.end put cur to result
    -----
            -------
    2 next.start <=cur.end:
    ---------------
       ------
    cur.end = max(cur.end, next.end)
    '''
    intervals.sort(key = lambda x : (x[0], x[1]))
    cur = intervals[0]
    res = []
    for i in range(1, len(intervals)):
        next = intervals[i]
        if next[0] > cur[1]:
            res.append(cur.copy())
            cur = next
        elif next[0] <= cur[1]:
            cur[1] = max(cur[1], next[1])
    if cur:
        res.append(cur.copy())
    return res


class Node:
    def __init__(self, key, val):
        self.key = key
        self.val = val
        self.prev, self.next = None, None

class LRUCache:

    def __init__(self, capacity: int):
        self.cap = capacity
        self.left = Node(0, 0)
        self.right = Node(0, 0)
        self.left.next = self.right
        self.right.prev = self.left
        self.cache = {}

    def remove(self, node):
        prev = node.prev
        next = node.next
        prev.next = next
        next.prev = prev

    def add(self, node):
        prev = self.right.prev
        prev.next = node
        self.right.prev = node
        node.next = self.right
        node.prev = prev

    def get(self, key: int) -> int:
        if key in self.cache:
            node = self.cache[key]
            self.remove(node)
            self.add(node)
            return node.val
        else:
            return -1

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.remove(self.cache[key])
        node = Node(key, value)
        self.cache[key] = node
        self.add(node)

        if len(self.cache) > self.cap:
            lru = self.left.next
            self.remove(lru)
            del self.cache[lru.key]


class AllOne {

    TreeMap<Integer, Set<String>> strByFreq = new TreeMap<>();
    Map<String, Integer> freqByStr = new HashMap<>();

    public void dec(String key) {
        int freq = freqByStr.get(key);
        freqByStr.put(key, freq - 1);
        if(freq==1) {
            freqByStr.remove(key);
        }
        strByFreq.get(freq).remove(key);
        if(strByFreq.get(freq).size() == 0) {
            strByFreq.remove(freq);
        }
        if(freq>1){
            strByFreq.putIfAbsent(freq-1, new HashSet<>());
            strByFreq.get(freq-1).add(key);
        }
    }

    public void inc(String key) {
        Integer freq = freqByStr.get(key);
        if(freq == null) {
            freqByStr.put(key, 1);
            strByFreq.putIfAbsent(1, new HashSet<>());
            strByFreq.get(1).add(key);
        } else {
            freqByStr.put(key, freq+1);
            //decrease old freq, increase new
            strByFreq.get(freq).remove(key);
            if (strByFreq.get(freq).size()==0) {
                strByFreq.remove(freq);
            }
            strByFreq.putIfAbsent(freq+1, new HashSet<>());
            strByFreq.get(freq+1).add(key);
        }
    }

    public String getMaxKey() {
        if(freqByStr.size()==0){
            return "";
        }
        return strByFreq.get(strByFreq.lastKey()).stream().findFirst().get();
    }

    public String getMinKey() {
        if(freqByStr.size()==0){
            return "";
        }
        return strByFreq.get(strByFreq.firstKey()).stream().findFirst().get();
    }
}


public List<String> fullJustify(String[] words, int maxWidth) {
    List<List<String>> lines = breakdown(words, maxWidth);

    List<String> res = new ArrayList<>();

    for(int i=0; i < lines.size() - 1; i++){
        List<String> line = lines.get(i);
        res.add( createLine(line, maxWidth, false) );
    }
    res.add( createLine(lines.get(lines.size()-1), maxWidth, true) );

    return res;
}

String createLine(
    List<String> line,
    int maxWidth,
    boolean isLast
) {
    ///////////////////////////  DONE  ///////////////////////////
    int baseLength = -1;
    for (String w : line) {
        baseLength += 1 + w.length();
    }
    int extraSpaces = maxWidth - baseLength;
    if (isLast || line.size()==1) {
        return String.join(" ", line) + " ".repeat(extraSpaces);
    }
    ///////////////////////////////////////////////////////
    int wordCount = line.size() - 1;
    // int spaceCount = maxWidth - wordCount;
    int spacesPerWord = extraSpaces / wordCount;
    int extraSpacesNeeded = extraSpaces % wordCount;

    for (int i=0; i< extraSpacesNeeded; i++) {
        line.set(i, line.get(i)+" ");
    }

    for (int i=0; i< wordCount; i++) {
        line.set(i, line.get(i)+" ".repeat(spacesPerWord));
    }
    return String.join(" ", line);
}

List<List<String>> breakdown(String[] words, int width) {
    List<List<String>> res = new ArrayList<>();
    List<String> curLine = new ArrayList<>();
    int curChars = 0;
    for (String w : words) {
        int charsIfAdding = curChars + curLine.size() + w.length();

        if (charsIfAdding > width) {
            res.add(curLine);
            curLine = new ArrayList<>();
            curChars = 0;
        }
        curChars += w.length();
        curLine.add(w);
    }
    if(curLine.size()>0){
        res.add(curLine);
    }
    return res;
}


class StockPrice {

    Map<Integer, Integer>  priceByTs = new ConcurrentHashMap<>();
    TreeMap<Integer, Integer> freqByPrice = new TreeMap<>();
    int maxTs = -1;

    public void update(int timestamp, int price) {
        maxTs = Math.max(maxTs, timestamp);

        Integer oldPrice = priceByTs.get(timestamp);
        if(oldPrice != null) {
            Integer freq = freqByPrice.get(oldPrice);
            if (freq == 1) {
                freqByPrice.remove(oldPrice);
            } else {
                freqByPrice.put(oldPrice, freq - 1);
            }
        }

        priceByTs.put(timestamp, price);

        Integer freq = freqByPrice.get(price);
        if (freq == null) {
            freqByPrice.put(price, 1);
        } else {
            freqByPrice.put(price, freq + 1);
        }
    }

    public int current() {
        return priceByTs.get(maxTs);
    }

    public int maximum() {
        return freqByPrice.lastKey();
    }

    public int minimum() {
        return freqByPrice.firstKey();
    }
  }


class Solution {
    public List<String> findHighAccessEmployees(List<List<String>> access_times) {
        /*
            get for 1 employee
            sort
            find out
        */
        Map<String, List<String>> timesByEmployee = new HashMap<>();
        for(List<String> rec : access_times) {
            String empl = rec.get(0);
            String time = rec.get(1);
            timesByEmployee.putIfAbsent(empl, new ArrayList<>());
            timesByEmployee.get(empl).add(time);
        }

        List<String> res = new ArrayList<>();

        for(var entry:timesByEmployee.entrySet()) {
            List<String> times = entry.getValue();
            Collections.sort(times);
            for(int i=1; i<times.size()-1;i++){
                if( diff(times.get(i-1), times.get(i)) + diff(times.get(i), times.get(i+1)) <60 ) {
                    res.add(entry.getKey());
                    break;
                }
            }
        }
        return res;
    }
    int diff(String a, String b) {
        int hoursA = Integer.valueOf(a.substring(0,2));
        int hoursB = Integer.valueOf(b.substring(0,2));

        int minsA = Integer.valueOf(a.substring(2));
        int minsB = Integer.valueOf(b.substring(2));
        int res = (hoursB - hoursA)*60;
        if (minsB < minsA){
            minsB += 60;
            res -= 60;
        }
        res += minsB - minsA;
        //System.out.println("between "+a+" and "+b+" " +res + " mins");
        return res;
    }
}


class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        ''' Approach:
        for each cell, if it is 1st letter ot the word:
            dfs
                pass: next letter to look for
                    set of visited
                    dfs returns what? True/False
        '''
        ROWS, COLS = len(board), len(board[0])

        def dfs(r,c, i, visited):
            if i == len(word):
                return True

            if r < 0 or r >= ROWS or c < 0 or c >= COLS or (r,c) in visited or word[i] != board[r][c]:
                return False

            visited.add((r,c))
            if (dfs(r+1,c,i+1,visited)
                or dfs(r-1,c,i+1,visited)
                or dfs(r,c+1, i+1, visited)
                or dfs(r,c-1, i+1, visited)):
                return True
            visited.remove((r,c))

        for r in range(ROWS):
            for c in range(COLS):
                if board[r][c] == word[0]:
                    if dfs(r, c, 0, set()):
                        return True
        return False


public Node lowestCommonAncestor(Node p, Node q) {
    /*
        get distance for both
        advance the lower one by the diff (so they're both at one level)
        advance node=node.parent until they point to the same node
    */
    Node pCopy = p;
    Node qCopy = q;

    int pLen = 0;
    while (p!=null) {
        p = p.parent;
        pLen++;
    }

    int qLen = 0;
    while (q!=null) {
        q = q.parent;
        qLen++;
    }

    p = pCopy;
    q = qCopy;
    while (pLen>qLen) {
        p = p.parent;
        pLen--;
    }
    while (qLen>pLen){
        q = q.parent;
        qLen--;
    }

    while(p!=q){
        p = p.parent;
        q = q.parent;
    }
    return p;

}
